; Regla 10 — Operador menor que en hora
user operator AND hour < 18

; Regla 11 — Operador mayor que
user admin AND hour > 7

; Regla 12 — Restricción de día domingo con punto y coma
user guest AND day = 'Sunday';

; Regla 13 — Recurso con extensión de ruta
user admin AND resource = 'etc/nginx/nginx.conf'

; Regla 14 — Operador <= con recurso
user operator AND hour <= 20 AND resource = 'reports.pdf'

; Regla 15 — Combinación NOT + AND
user admin AND NOT resource = 'root.key' AND hour >= 9
